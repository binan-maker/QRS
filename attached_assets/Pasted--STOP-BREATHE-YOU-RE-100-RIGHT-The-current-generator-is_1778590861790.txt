### Day 2: Build Custom QR Flow
**New file:** `app/generate/custom.tsx`
```tsx
import { useState } from 'react';
import { View, Text, TextInput, TouchableOpacity, ScrollView, StyleSheet } from 'react-native';
import { router } from 'expo-router';
import { Ionicons } from '@expo/vector-icons';
import { doc, setDoc, collection } from 'firebase/firestore';
import { auth, db } from '@/lib/firebase';

export default function CreateCustomQR() {
  const [name, setName] = useState('');
  const [fields, setFields] = useState([{ key: '', label: '', type: 'text' }]);
  const [template, setTemplate] = useState('');

  const addField = () => setFields([...fields, { key: '', label: '', type: 'text' }]);
  
  const handleSave = async () => {
    if (!name || !template) return;
    const uid = auth.currentUser?.uid;
    if (!uid) return;

    const customId = `custom_${Date.now()}`;
    await setDoc(doc(db, 'users', uid, 'customTemplates', customId), {
      name,
      fields,
      template,
      createdAt: new Date(),
      isPrivate: true
    });

    router.replace(`/generate/preview?templateId=${customId}`);
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>Create Custom QR</Text>
      
      <TextInput 
        style={styles.input} 
        placeholder="Name (e.g., WhatsApp Business)" 
        value={name} 
        onChangeText={setName} 
      />

      {fields.map((field, idx) => (
        <View key={idx} style={styles.fieldRow}>
          <TextInput style={styles.inputSmall} placeholder="Field key (e.g., phone)" value={field.key} onChangeText={v => {
            const newFields = [...fields]; newFields[idx].key = v; setFields(newFields);
          }} />
          <TextInput style={styles.inputSmall} placeholder="Label (e.g., Phone Number)" value={field.label} onChangeText={v => {
            const newFields = [...fields]; newFields[idx].label = v; setFields(newFields);
          }} />
        </View>
      ))}

      <TouchableOpacity onPress={addField} style={styles.addButton}>
        <Ionicons name="add-circle" size={20} color="#3b82f6" />
        <Text style={styles.addText}>Add Another Field</Text>
      </TouchableOpacity>

      <Text style={styles.sectionTitle}>Output Template</Text>
      <TextInput 
        style={[styles.input, styles.textArea]} 
        placeholder="e.g., https://wa.me/{{phone}}&text={{message}}" 
        value={template} 
        onChangeText={setTemplate} 
        multiline 
      />
      <Text style={styles.hint}>Use {{key}} for dynamic fields</Text>

      <TouchableOpacity style={styles.saveButton} onPress={handleSave}>
        <Text style={styles.saveText}>Save & Generate QR</Text>
      </TouchableOpacity>
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, backgroundColor: '#fff' },
  title: { fontSize: 20, fontWeight: '700', marginBottom: 20 },
  input: { borderWidth: 1, borderColor: '#cbd5e1', borderRadius: 10, padding: 14, marginBottom: 16, fontSize: 15 },
  fieldRow: { flexDirection: 'row', gap: 12, marginBottom: 12 },
  inputSmall: { flex: 1, borderWidth: 1, borderColor: '#cbd5e1', borderRadius: 8, padding: 12, fontSize: 14 },
  addButton: { flexDirection: 'row', alignItems: 'center', gap: 8, marginBottom: 24 },
  addText: { color: '#3b82f6', fontWeight: '500' },
  sectionTitle: { fontSize: 16, fontWeight: '600', marginBottom: 8 },
  textArea: { minHeight: 80, textAlignVertical: 'top' },
  hint: { fontSize: 12, color: '#64748b', marginBottom: 20 },
  saveButton: { backgroundColor: '#3b82f6', padding: 16, borderRadius: 12, alignItems: 'center' },
  saveText: { color: '#fff', fontSize: 16, fontWeight: '600' }
});
```
