## 🔗 1. Cloud Function: `/go/:slug` Redirect Engine
This is the core that makes "edit destination without reprinting QR" possible. It intercepts scans, validates status, logs analytics, and redirects to the current destination.

```typescript
// functions/src/redirect.ts
import * as functions from 'firebase-functions';
import * as admin from 'firebase-admin';
import { increment } from 'firebase-admin/firestore';

admin.initializeApp();

export const handleQrRedirect = functions.https.onRequest(async (req, res) => {
  // Extract slug: /go/abc123 or /go/abc123?utm=...
  const pathParts = req.path.split('/').filter(Boolean);
  const slug = pathParts[pathParts.length - 1];

  if (!slug || slug.length < 5) {
    return res.status(400).send('Invalid QR code');
  }

  try {
    // 🔍 Fast lookup (ensure redirectSlug is indexed in Firestore)
    const qrSnap = await admin.firestore()
      .collection('qrCodes')
      .where('redirectSlug', '==', slug)
      .limit(1)
      .get();

    if (qrSnap.empty) {
      return res.status(404).send('QR code not found or has been deleted');
    }

    const qrDoc = qrSnap.docs[0];
    const data = qrDoc.data();

    // 🚦 Status checks
    if (data.status === 'paused') {
      return res.status(403).send('This QR code is temporarily paused by the owner');
    }
    if (data.status === 'archived') {
      return res.status(410).send('This QR code has been archived');
    }

    // 🎯 Resolve destination
    const destination = data.destinationUrl || data.rawContent;
    if (!destination) {
      return res.status(500).send('No destination configured');
    }

    // 📊 Non-blocking scan analytics
    admin.firestore().collection('qrCodes').doc(qrDoc.id).update({
      scanCount: increment(1),
      lastScannedAt: admin.firestore.FieldValue.serverTimestamp()
    }).catch(err => console.error('Scan count update failed:', err));

    // 🔄 Redirect (302 allows future changes to take effect immediately)
    return res.redirect(302, destination);
  } catch (error) {
    console.error('Redirect error:', error);
    return res.status(500).send('Internal server error');
  }
});
```

**Deploy:**
```bash
firebase deploy --only functions:handleQrRedirect
```
**Route in `firebase.json`:**
```json
"hosting": {
  "rewrites": [
    { "source": "/go/**", "function": "handleQrRedirect" }
  ]
}
```

## 🔒 2. Enhanced Security Scanner (Unified for Saved & Business)
Replace/augment your current validation with this production-ready pipeline. It runs **before any destination change is saved** and works identically for both Saved and Business QRs.

```typescript
// lib/security/url-scanner.ts
import axios from 'axios';

const SAFE_BROWSING_API_KEY = process.env.EXPO_PUBLIC_SAFE_BROWSING_KEY;
const BLOCKED_SHORTENERS = ['bit.ly', 'tinyurl.com', 't.co', 'short.link', 'cutt.ly', 'rb.gy'];
const PHISHING_KEYWORDS = ['verify-account', 'secure-login', 'update-payment', 'apple-id', 'netflix-billing', 'wallet-confirm'];

export async function validateAndScanUrl(url: string): Promise<{ 
  valid: boolean; 
  error?: string; 
  threat?: string;
}> {
  // 1️⃣ Format validation
  try { new URL(url); } catch { return { valid: false, error: 'Invalid URL format' }; }
  const parsed = new URL(url);

  // 2️⃣ Block raw IPs
  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(parsed.hostname)) {
    return { valid: false, error: 'IP addresses are blocked for security reasons' };
  }

  // 3️⃣ Block private/local networks
  const privateHosts = ['localhost', '127.0.0.1', '0.0.0.0', '::1'];
  if (privateHosts.includes(parsed.hostname) || parsed.hostname.endsWith('.local') || parsed.hostname.startsWith('192.168.') || parsed.hostname.startsWith('10.')) {
    return { valid: false, error: 'Private/local network addresses are not allowed' };
  }

  // 4️⃣ Block URL shorteners (prevents hidden redirects)
  if (BLOCKED_SHORTENERS.some(s => parsed.hostname.includes(s))) {
    return { valid: false, error: 'URL shorteners are blocked to prevent hidden redirects' };
  }

  // 5️⃣ Keyword-based phishing/typosquatting detection
  const fullPath = (parsed.hostname + parsed.pathname).toLowerCase();
  if (PHISHING_KEYWORDS.some(kw => fullPath.includes(kw))) {
    return { valid: false, error: 'URL contains suspicious patterns commonly used in phishing' };
  }

  // 6️⃣ Google Safe Browsing API (optional but recommended)
  if (SAFE_BROWSING_API_KEY) {
    try {
      const response = await axios.post(
        `https://safebrowsing.googleapis.com/v4/threatMatches:find?key=${SAFE_BROWSING_API_KEY}`,
        {
          client: { clientId: 'qrguard-app', clientVersion: '1.0.0' },
          threatInfo: {
            threatTypes: ['MALWARE', 'SOCIAL_ENGINEERING', 'UNWANTED_SOFTWARE', 'POTENTIALLY_HARMFUL_APPLICATION'],
            platformTypes: ['ANY_PLATFORM'],
            threatEntryTypes: ['URL'],
            threatEntries: [{ url }]
          }
        }
      );
      if (response.data.matches?.length > 0) {
        return { 
          valid: false, 
          error: 'URL flagged as unsafe by Google Safe Browsing', 
          threat: response.data.matches[0].threatType 
        };
      }
    } catch (apiErr) {
      console.warn('Safe Browsing API unavailable, falling back to local rules');
      // Don't block saves on API failure, but log for monitoring
    }
  }

  return { valid: true };
}
```

## 🔌 How to Wire Into Your Existing Edit Flow

In your `EditDestinationModal` (used by both Saved & Business):

```typescript
import { validateAndScanUrl } from '@/lib/security/url-scanner';

const handleSaveDestination = async (newUrl: string) => {
  setIsValidating(true);
  
  // 🔒 Run unified security scan
  const scan = await validateAndScanUrl(newUrl);
  if (!scan.valid) {
    setInlineError(scan.error);
    setIsValidating(false);
    return;
  }

  setInlineError(null);
  setShowConfirmModal(true); // Your existing confirmation step
  setIsValidating(false);
};

const handleConfirmChange = async () => {
  try {
    const qrRef = doc(db, 'qrCodes', qrId);
    await updateDoc(qrRef, {
      destinationUrl: newUrl,
      updatedAt: serverTimestamp(),
      contentChangeLog: arrayUnion({
        changedAt: serverTimestamp(),
        from: currentDestination,
        to: newUrl,
        changedBy: auth.currentUser?.uid
      })
    });
    
    // Trim log to 10 entries (client-side or via Cloud Function trigger)
    showToast('Destination updated successfully', 'checkmark-circle');
    setShowConfirmModal(false);
  } catch (err) {
    showToast('Failed to update destination', 'alert-circle');
  }
};
```

**Key Insight:** The architecture is now **unified**. Saved and Business only differ in *management depth* (analytics, team roles, branding), not in core routing or security. This reduces bugs, simplifies testing, and scales cleanly.

## 🚀 Deployment Steps

1. **Add Safe Browsing Key** (optional but recommended):
   - Go to [Google Cloud Console](https://console.cloud.google.com/) → APIs → Safe Browsing API → Enable
   - Create API key → Add to `.env`: `EXPO_PUBLIC_SAFE_BROWSING_KEY=your_key`
2. **Deploy Cloud Function**:
   ```bash
   cd functions && npm install axios firebase-admin firebase-functions
   firebase deploy --only functions:handleQrRedirect,hosting
   ```
3. **Index `redirectSlug`** in Firestore Console for fast lookups
4. **Test Flow**:
   - Create Saved QR → Edit destination → Validate → Confirm → Scan `/go/:slug` → Redirects correctly
   - Repeat for Business QR → Same behavior, same security
