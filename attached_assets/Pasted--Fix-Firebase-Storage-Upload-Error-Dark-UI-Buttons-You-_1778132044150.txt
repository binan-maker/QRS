## 🔥 Complete Upload Flow Fix

Update your storage service to handle errors gracefully:

```typescript
// lib/services/storage-service.ts
import { getStorage, ref, uploadBytes, getDownloadURL } from 'firebase/storage';
import { auth } from '@/lib/firebase';

const storage = getStorage();

export async function uploadProfilePhoto(imageUri: string) {
  try {
    const user = auth.currentUser;
    if (!user) {
      throw new Error('User not authenticated');
    }

    // Fetch the image
    const response = await fetch(imageUri);
    const blob = await response.blob();

    // Create unique filename
    const timestamp = Date.now();
    const randomId = Math.random().toString(36).substring(7);
    const fileName = `profile-photos/${user.uid}_${timestamp}_${randomId}.jpeg`;

    // Create storage reference
    const storageRef = ref(storage, fileName);

    // Upload with metadata
    const metadata = {
      contentType: 'image/jpeg',
      customMetadata: {
        uploadedBy: user.uid,
        uploadedAt: new Date().toISOString(),
      },
    };

    // Upload file
    const snapshot = await uploadBytes(storageRef, blob, metadata);

    // Get download URL
    const downloadURL = await getDownloadURL(snapshot.ref);

    return {
      success: true,
      url: downloadURL,
      path: fileName,
    };
  } catch (error: any) {
    console.error('[storage] uploadProfilePhoto failed:', error);
    
    // Handle specific errors
    if (error.code === 'storage/unauthorized') {
      return {
        success: false,
        error: 'Permission denied. Please check your Firebase Storage rules.',
      };
    }
    
    return {
      success: false,
      error: error.message || 'Failed to upload image',
    };
  }
}
```

service firebase.storage {
  match /b/{bucket}/o {
    
    // 🟢 Profile Photos: userId subfolder for isolation
    match /profile-photos/{userId}/{fileName} {
      allow read: if true; // Public read for profile display
      allow write: if request.auth != null 
                    && request.auth.uid == userId
                    && fileName.matches('^[a-zA-Z0-9_-]+\\.(jpg|jpeg|png|webp)$'); // Validate filename
    }
    
    // 🟢 QR Logos: Same structure
    match /qr-logos/{userId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null 
                    && request.auth.uid == userId
                    && request.resource.size < 5 * 1024 * 1024 // Max 5MB
                    && fileName.matches('^[a-zA-Z0-9_-]+\\.(jpg|jpeg|png|webp|svg)$');
    }
    
    // 🟢 Verification Docs: More restrictive
    match /verification-docs/{userId}/{fileName} {
      allow read: if request.auth != null 
                   && (request.auth.uid == userId || isModerator());
      allow write: if request.auth != null 
                    && request.auth.uid == userId
                    && request.resource.size < 10 * 1024 * 1024 // Max 10MB
                    && fileName.matches('^[a-zA-Z0-9_-]+\\.(jpg|jpeg|png|pdf)$');
    }
    
    // 🟢 Public assets (app icons, etc.)
    match /public/{fileName} {
      allow read: if true;
      allow write: if false; // Only upload via Firebase Console/CLI
    }
    
    // 🔴 Default deny everything else
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
  
  // Helper function for role-based access (optional)
  function isModerator() {
    return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'moderator';
  }
}
