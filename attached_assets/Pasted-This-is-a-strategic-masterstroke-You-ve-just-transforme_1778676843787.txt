// lib/services/generator-service.ts
export async function generateQRPayload(mode: 'standard' | 'business' | 'private', content: string) {
  if (mode === 'private') return content; // Raw, offline, zero trace
  
  const slug = generateSecureSlug(6); // 'a7x9k2'
  await setDoc(doc(db, 'qrRegistry', slug), {
    destination: content,
    mode,
    ownerId: auth.currentUser?.uid,
    createdAt: serverTimestamp(),
    status: 'active',
    threatScore: await aiScanContent(content) // Security-first default
  });
  
  return `https://guard.link/${slug}`; // Standard/Business always route through your DB
}

### 3. Build the Resolution API (Your B2B Revenue Core)
```typescript
// functions/src/resolve.ts
export const resolveQR = functions.https.onRequest(async (req, res) => {
  const slug = req.path.split('/').pop();
  const record = await getDoc(doc(db, 'qrRegistry', slug));
  
  if (!record.exists() || record.data().status !== 'active') {
    return res.redirect(302, 'https://qrguard.in/qr-not-found');
  }

  // Log scan (non-blocking, analytics pipeline)
  logScanEvent(slug, req.headers, req.ip).catch(console.error);

  // Return JSON for API consumers, redirect for browsers
  if (req.headers['accept']?.includes('application/json')) {
    return res.json({
      destination: record.data().destination,
      trustScore: record.data().threatScore,
      mode: record.data().mode,
      verified: record.data().ownerVerified || false
    });
  }
  
  // Browser fallback: show security badge + proceed
  return res.redirect(302, `https://qrguard.in/verify/${slug}?next=${encodeURIComponent(record.data().destination)}`);
});
```
