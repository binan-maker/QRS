|-------|--------------|------|
| **Dual-Write Pattern** | `saveGeneratedQr()` writes to `users/{uid}/generatedQrs` AND `qrCodes/{qrId}` | Data inconsistency if one write fails; 2x write cost |
| **Half-Baked Embedded Stats** | `scanCount` embedded but falls back to `qrCodes` if `=== 0` | Unpredictable read patterns; hard to cache |

| **Synchronous Crypto Operations** | `Crypto.digestStringAsync()` in main thread | UI jank on low-end Android; poor perceived performance |
| **No Pagination/Cursors** | `getUserGeneratedQrs()` fetches everything at once | Slow initial load; OOM crashes at 100+ QRs/user |
| **Silent Error Handling** | `catch {}` blocks return `null` without logging | Production debugging impossible; user trust erosion |

```typescript
// New generation flow
const generateQR = async (content: string, options: GenerateOptions) => {
  // ✅ Default: wrap in secure redirect
  const redirectSlug = await createGuardLink(content, options.securityLevel);
  const qrPayload = `https://qrguard.in/r/${redirectSlug}`;
  
  // ✅ Show threat score during generation
  const threatScan = await aiScanContent(content);
  if (threatScan.riskScore > 50) {
    showWarning(`⚠️ This content has ${threatScan.riskScore}% risk score`);
  }
  
  // ✅ Only allow raw QR if user explicitly opts in + acknowledges risk
  if (options.allowRaw && threatScan.riskScore < 20) {
    return generateRawQR(content);
  }
  
  return generateQR(qrPayload); // Secure by default
};
```

### Fix 1: Add Pagination to `getUserGeneratedQrs`
```typescript
// Before: Loads all QRs at once

// After: Pagination support
export async function getUserGeneratedQrs(
  userId: string, 
  options: { limit?: number; startAfter?: any } = {}
): Promise<{ items: GeneratedQrItem[]; nextCursor?: any }> {
  const { limit = 25, startAfter } = options;
  
  const baseQuery = query(
    collection(db, 'users', userId, 'generatedQrs'),
    orderBy('createdAt', 'desc'),
    limit(limit),
    ...(startAfter ? [startAfter(startAfter)] : [])
  );
  
  const snap = await getDocs(baseQuery);
  return {
    items: snap.docs.map(...), // field projection
    nextCursor: snap.docs.length === limit ? snap.docs[snap.docs.length - 1] : null
  };
}
```

### Fix 2: Debounce Real-Time Subscription
```typescript
// features/generator/hooks/useQrSubscription.ts
import { debounce } from 'lodash';

export function useQrSubscription(userId: string, onUpdate: (items) => void) {
  // ✅ Debounce updates to prevent 50 rapid re-renders
  const debouncedUpdate = useMemo(
    () => debounce(onUpdate, 300, { leading: false, trailing: true }),
    [onUpdate]
  );
  
  useEffect(() => {
    const unsub = subscribeToUserGeneratedQrs(userId, debouncedUpdate);
    return () => {
      unsub();
      debouncedUpdate.cancel(); // ✅ Clean up debounce timer
    };
  }, [userId, debouncedUpdate]);
}
```

```typescript
import { Platform } from 'react-native';
import { runOnBackground } from 'react-native-background-task';

export async function generateSignatureAsync(content: string, userId: string): Promise<string> {
  // ✅ Offload to native/background thread to avoid UI jank
  if (Platform.OS === 'android') {
    return await runOnBackground(() => 
      Crypto.digestStringAsync(Crypto.CryptoDigestAlgorithm.SHA256, content + '|' + userId + '|' + SALT)
    );
  }
  // Fallback for iOS/web
  return await Crypto.digestStringAsync(...);
}
```
