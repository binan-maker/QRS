### Step 1: Cache Profile Data Aggressively (Cut Reads by 95%)

```typescript
// lib/hooks/useProfileData.ts
import { useQuery } from '@tanstack/react-query';
import { doc, getDoc } from 'firebase/firestore';
import { db } from '@/lib/firebase';
import { mmkv } from '@/lib/storage/mmkv'; // Fast local storage

const CACHE_KEY = 'profile_data_v1';

export function useProfileData(uid: string) {
  return useQuery({
    queryKey: ['profile', uid],
    queryFn: async () => {
      // ✅ Field projection: fetch ONLY what UI needs
      const snap = await getDoc(doc(db, 'users', uid));
      const d = snap.data() || {};
      
      const data = {
        displayName: d.displayName || 'User',
        email: d.email || '',
        avatarUrl: d.avatarUrl || null,
        avatarVersion: d.avatarVersion || 0, // For cache busting
        stats: {
          scans: d.totalScans || 0, // Pre-aggregated in Firestore
          created: d.generatedCount || 0,
          followers: d.followers || 0
        }
      };

      // ✅ Cache to MMKV for instant load on next visit
      mmkv.set(`profile_${uid}`, JSON.stringify(data));
      return data;
    },
    // ✅ Load from cache FIRST (0ms perceived load)
    initialData: () => {
      const raw = mmkv.getString(`profile_${uid}`);
      return raw ? JSON.parse(raw) : undefined;
    },
    staleTime: 1000 * 60 * 5,   // 5 min cache
    gcTime: 1000 * 60 * 30,     // Keep in memory 30 min
    refetchOnWindowFocus: false, // ✅ Stop re-fetch on tab switch
    refetchOnMount: false,       // ✅ Critical: no re-fetch on navigation
  });
}
```

## ✅ Solution Architecture: Fast Cards + Cached Queries + Real-Time Scan Count

```
App Mount → React Query Cache → FlashList (virtualized) → CachedAvatar (expo-image)
                              ↑
Pull-to-Refresh → Invalidate Cache → Re-fetch → Update Scan Counts
                              ↑
Real-Time Scan Count → Optional onSnapshot on scanCount field ONLY (not full doc)
```

**Key Principles:**
1. **Virtualize List**: `FlashList` with `estimatedItemSize` → O(visible) memory
2. **Cache Queries**: React Query + MMKV → 0 reads on tab switch
3. **Field Projection**: Fetch ONLY `id`, `qrType`, `preview`, `scanCount`, `thumbnailUrl`
4. **Scan Count Strategy**: Cache everything EXCEPT `scanCount` → real-time via lightweight listener OR pull-to-refresh
5. **Image Handling**: `expo-image` with `cachePolicy="memory-disk"` + placeholder + error fallback

### 1️⃣ Cached Query Hook with Selective Real-Time Scan Count

```typescript
// lib/hooks/useMyQrs.ts
import { useQuery, useQueryClient } from '@tanstack/react-query';
import { collection, query, orderBy, limit, getDocs, doc, onSnapshot } from 'firebase/firestore';
import { db } from '@/lib/firebase';
import { mmkv } from '@/lib/storage/mmkv';

const CACHE_KEY = 'my_qrs_v1';

// ✅ Fetch paginated list with field projection
async function fetchMyQrsPage(uid: string, pageParam?: any) {
  const baseQuery = query(
    collection(db, 'users', uid, 'generatedQrs'),
    orderBy('createdAt', 'desc'),
    limit(25),
    ...(pageParam ? [startAfter(pageParam)] : [])
  );

  const snap = await getDocs(baseQuery);
  const items = snap.docs.map(doc => {
    const d = doc.data();
    return {
      id: doc.id,
      type: d.qrType || 'unknown',
      preview: d.content?.slice(0, 80) || '',
      scanCount: d.scanCount || 0, // ✅ This field can be real-time
      thumbnailUrl: d.thumbnailUrl || null,
      createdAt: d.createdAt?.toDate?.() || new Date(),
      status: d.status || 'active'
    };
  });

  return {
    items,
    nextCursor: snap.docs.length === 25 ? snap.docs[snap.docs.length - 1] : null,
  };
}

export function useMyQrs(uid: string) {
  const queryClient = useQueryClient();

  const { data, fetchNextPage, hasNextPage, isFetchingNextPage, refetch } = useInfiniteQuery({
    queryKey: ['myQrs', uid],
    queryFn: ({ pageParam }) => fetchMyQrsPage(uid, pageParam),
    getNextPageParam: (lastPage) => lastPage.nextCursor,
    initialPageParam: null,
    initialData: () => {
      const raw = mmkv.getString(CACHE_KEY);
      return raw ? JSON.parse(raw) : undefined;
    },
    staleTime: 1000 * 60 * 5,   // 5 min cache for list structure
    gcTime: 1000 * 60 * 30,     // 30 min memory
    refetchOnWindowFocus: false,
    refetchOnMount: false,
  });

  // ✅ Optional: Real-time scan count updates ONLY (lightweight)
  const subscribeToScanCounts = useCallback(() => {
    const items = data?.pages.flatMap(p => p.items) || [];
    const unsubscribes = items.map(item => {
      return onSnapshot(doc(db, 'users', uid, 'generatedQrs', item.id), (snap) => {
        const newCount = snap.data()?.scanCount || 0;
        if (newCount !== item.scanCount) {
          // ✅ Update cache only for scanCount field
          queryClient.setQueryData(['myQrs', uid], (old: any) => {
            if (!old) return old;
            return {
              ...old,
              pages: old.pages.map((page: any) => ({
                ...page,
                items: page.items.map((i: any) => 
                  i.id === item.id ? { ...i, scanCount: newCount } : i
                )
              }))
            };
          });
        }
      });
    });
    return () => unsubscribes.forEach(unsub => unsub());
  }, [data, uid, queryClient]);

  return {
    data,
    fetchNextPage,
    hasNextPage,
    isFetchingNextPage,
    refetch, // ✅ Use this for pull-to-refresh
    subscribeToScanCounts, // ✅ Call this if you want real-time counts
  };
}
```

### 4️⃣ Pull-to-Refresh + Real-Time Scan Count Strategy

**Option A: Pull-to-Refresh Only (Recommended for Cost)**
- Set `staleTime: 5min` in `useMyQrs`
- User pulls to refresh → `refetch()` → updates scan counts
- Zero background listeners → minimal Firestore cost

**Option B: Lightweight Real-Time Scan Counts**
- Use `subscribeToScanCounts()` from hook above
- Only listens to `scanCount` field changes (not full doc)
- Cost: ~1 read/scan event (vs. 20+ for full doc listener)

**Recommendation**: Start with **Option A** (pull-to-refresh). Add Option B later if users demand instant count updates.
