```typescript
export interface CategorySchema {
  id: string;                    // Unique ID: 'upi_payment', 'whatsapp_business'
  name: string;                  // Human-readable: 'UPI Payment QR'
  description: string;           // "Accept payments via GPay, PhonePe, Paytm"

  region?: 'global' | 'india' | 'us' | 'eu'; // Geo-targeting
  
  // Input fields definition (auto-generates form)
  fields: FieldDefinition[];
  
  // Validation logic (can be regex, function ref, or AI prompt)
  validation: {
    type: 'regex' | 'function' | 'ai';
    rule: string; // Regex pattern or function ID or AI prompt
  };
  
  // Output format (how to construct final QR content)
  output: {
    template: string; // "upi://pay?pa={{vpa}}&am={{amount}}&tn={{note}}"
    encoding?: 'plain' | 'base64' | 'json';
  };
  
  // Security & trust settings
  security: {
    requiresVerification?: boolean;
    threatScan?: boolean; // Run AI scan before generation
    guardLinkDefault?: boolean; // Wrap in guard.link by default
  };
  
  // Metadata for discovery
  tags: string[]; // ['payment', 'india', 'upi', 'merchant']
  popularity?: number; // For ranking in search
  createdAt: Timestamp;
  updatedAt: Timestamp;
}

export interface FieldDefinition {
  key: string;           // 'vpa', 'amount', 'note'
  label: string;         // 'UPI ID', 'Amount (₹)', 'Note'
  type: 'text' | 'number' | 'email' | 'phone' | 'url' | 'select' | 'date';
  required?: boolean;
  placeholder?: string;
  validation?: {
    regex?: string;
    min?: number;
    max?: number;
    pattern?: string;
  };
  options?: { label: string; value: string }[]; // For 'select' type
  autoDetect?: {
    from: 'clipboard' | 'camera' | 'contact';
    transform?: string; // Function to extract value
  };
}
```

### Step 2: Build Category Registry Service (Firestore + Cache)

```typescript
// lib/services/CategoryRegistryService.ts
import { collection, query, where, getDocs, doc, getDoc } from 'firebase/firestore';
import { db } from '@/lib/firebase';
import { mmkv } from '@/lib/storage/mmkv';

const CACHE_KEY = 'category_registry_v1';

export class CategoryRegistryService {
  // ✅ Fetch categories with caching + geo-filtering
  static async getCategories(options: { 
    region?: string; 
    search?: string; 
    tags?: string[] 
  } = {}) {
    const cacheKey = `${CACHE_KEY}_${options.region || 'global'}_${options.search || ''}`;
    const cached = mmkv.getString(cacheKey);
    if (cached) return JSON.parse(cached);

    let q = query(collection(db, 'categories'));
    
    if (options.region) {
      q = query(q, where('region', 'in', ['global', options.region]));
    }
    if (options.tags?.length) {
      q = query(q, where('tags', 'array-contains-any', options.tags.slice(0, 10)));
    }
    
    const snap = await getDocs(q);
    const categories = snap.docs.map(d => ({ id: d.id, ...d.data() }));
    
    // ✅ AI-powered search fallback (if text search provided)
    if (options.search) {
      const filtered = await this.aiFilterCategories(categories, options.search);
      mmkv.set(cacheKey, JSON.stringify(filtered));
      return filtered;
    }
    
    mmkv.set(cacheKey, JSON.stringify(categories));
    return categories;
  }
  
  // ✅ Get single category by ID (with cache)
  static async getCategory(categoryId: string) {
    const cacheKey = `category_${categoryId}`;
    const cached = mmkv.getString(cacheKey);
    if (cached) return JSON.parse(cached);
    
    const snap = await getDoc(doc(db, 'categories', categoryId));
    if (!snap.exists()) return null;
    
    const category = { id: snap.id, ...snap.data() };
    mmkv.set(cacheKey, JSON.stringify(category));
    return category;
  }
  
  // ✅ AI-powered category discovery (natural language → category)
  static async aiFilterCategories(categories: any[], query: string) {
    // Lightweight client-side fuzzy match first
    const lowerQuery = query.toLowerCase();
    const fuzzy = categories.filter(c => 
      c.name.toLowerCase().includes(lowerQuery) ||
      c.description.toLowerCase().includes(lowerQuery) ||
      c.tags.some((t: string) => t.includes(lowerQuery))
    );
    
    if (fuzzy.length > 0) return fuzzy;
    
    // Fallback: call AI endpoint for semantic matching
    try {
      const response = await fetch('/api/ai/match-category', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ query, categoryIds: categories.map(c => c.id) })
      });
      const { matches } = await response.json();
      return categories.filter(c => matches.includes(c.id));
    } catch {
      return categories.slice(0, 10); // Fallback to top 10
    }
  }
  
  // ✅ Register new category (admin/verified users only)
  static async registerCategory(schema: CategorySchema, userId: string) {
    // Validate schema
    if (!this.validateSchema(schema)) {
      throw new Error('Invalid category schema');
    }
    
    // Check permissions (only verified merchants or admins)
    const userDoc = await getDoc(doc(db, 'users', userId));
    if (!userDoc.data()?.verified && !userDoc.data()?.isAdmin) {
      throw new Error('Only verified users can register categories');
    }
    
    // Save to Firestore
    const ref = doc(collection(db, 'categories'));
    await setDoc(ref, {
      ...schema,
      createdAt: new Date(),
      updatedAt: new Date(),
      createdBy: userId,
      status: 'pending' // Requires admin approval for public visibility
    });
    
    return ref.id;
  }
  
  static validateSchema(schema: CategorySchema): boolean {
    // Basic validation: required fields, valid field types, etc.
    return !!(schema.id && schema.name && schema.fields?.length > 0 && schema.output?.template);
  }
}
```

### Step 3: Dynamic Form Renderer (Auto-Generate UI from Schema)

```tsx
// features/generator/components/DynamicCategoryForm.tsx
import React, { useMemo } from 'react';
import { View, Text, TextInput, TouchableOpacity, StyleSheet } from 'react-native';
import { Controller, useForm } from 'react-hook-form';
import { CategorySchema, FieldDefinition } from '@/lib/schemas/CategorySchema';

interface DynamicCategoryFormProps {
  category: CategorySchema;
  onSubmit: (data: Record<string, any>) => void;
  defaultValues?: Record<string, any>;
}

export const DynamicCategoryForm: React.FC<DynamicCategoryFormProps> = ({
  category,
  onSubmit,
  defaultValues = {}
}) => {
  const { control, handleSubmit, formState: { errors } } = useForm({
    defaultValues: category.fields.reduce((acc, field) => {
      acc[field.key] = defaultValues[field.key] || field.placeholder || '';
      return acc;
    }, {} as Record<string, any>)
  });

  // ✅ Render field based on type
  const renderField = useMemo(() => (field: FieldDefinition) => {
    switch (field.type) {
      case 'select':
        return (
          <Controller
            name={field.key}
            control={control}
            rules={{ required: field.required ? `${field.label} is required` : false }}
            render={({ field: { onChange, value } }) => (
              <View style={styles.fieldContainer}>
                <Text style={styles.label}>{field.label}{field.required && '*'}</Text>
                <View style={styles.selectContainer}>
                  {field.options?.map(opt => (
                    <TouchableOpacity
                      key={opt.value}
                      style={[
                        styles.selectOption,
                        value === opt.value && styles.selectOptionSelected
                      ]}
                      onPress={() => onChange(opt.value)}
                    >
                      <Text style={[
                        styles.selectOptionText,
                        value === opt.value && styles.selectOptionTextSelected
                      ]}>
                        {opt.label}
                      </Text>
                    </TouchableOpacity>
                  ))}
                </View>
                {errors[field.key] && <Text style={styles.error}>{errors[field.key]?.message as string}</Text>}
              </View>
            )}
          />
        );
      
      case 'number':
        return (
          <Controller
            name={field.key}
            control={control}
            rules={{
              required: field.required ? `${field.label} is required` : false,
              pattern: field.validation?.regex ? new RegExp(field.validation.regex) : undefined,
              min: field.validation?.min,
              max: field.validation?.max
            }}
            render={({ field: { onChange, value } }) => (
              <View style={styles.fieldContainer}>
                <Text style={styles.label}>{field.label}{field.required && '*'}</Text>
                <TextInput
                  style={styles.input}
                  value={value}
                  onChangeText={onChange}
                  keyboardType="numeric"
                  placeholder={field.placeholder}
                />
                {errors[field.key] && <Text style={styles.error}>{errors[field.key]?.message as string}</Text>}
              </View>
            )}
          />
        );
      
      default: // text, email, phone, url
        return (
          <Controller
            name={field.key}
            control={control}
            rules={{
              required: field.required ? `${field.label} is required` : false,
              pattern: field.validation?.regex ? new RegExp(field.validation.regex) : undefined
            }}
            render={({ field: { onChange, value } }) => (
              <View style={styles.fieldContainer}>
                <Text style={styles.label}>{field.label}{field.required && '*'}</Text>
                <TextInput
                  style={styles.input}
                  value={value}
                  onChangeText={onChange}
                  keyboardType={getKeyboardType(field.type)}
                  autoCapitalize="none"
                  placeholder={field.placeholder}
                />
                {errors[field.key] && <Text style={styles.error}>{errors[field.key]?.message as string}</Text>}
              </View>
            )}
          />
        );
    }
  }, [control, errors]);

  const getKeyboardType = (type: string) => {
    switch (type) {
      case 'email': return 'email-address';
      case 'phone': return 'phone-pad';
      case 'url': return 'url';
      case 'number': return 'numeric';
      default: return 'default';
    }
  };

  return (
    <View style={styles.container}>
      {category.fields.map(field => (
        <React.Fragment key={field.key}>
          {renderField(field)}
        </React.Fragment>
      ))}
      
      <TouchableOpacity 
        style={styles.submitButton}
        onPress={handleSubmit(onSubmit)}
      >
        <Text style={styles.submitButtonText}>Generate {category.name}</Text>
      </TouchableOpacity>
    </View>
  );
};

const styles = StyleSheet.create({
  container: { padding: 16, gap: 16 },
  fieldContainer: { gap: 6 },
  label: { fontSize: 14, fontWeight: '500', color: '#1e293b' },
  input: {
    borderWidth: 1,
    borderColor: '#cbd5e1',
    borderRadius: 8,
    padding: 12,
    fontSize: 14,
    backgroundColor: '#f8fafc'
  },
  selectContainer: { flexDirection: 'row', flexWrap: 'wrap', gap: 8 },
  selectOption: {
    paddingHorizontal: 12,

    borderRadius: 6,
    backgroundColor: '#f1f5f9',
```

### Step 4: AI-Powered Category Discovery ("I need a QR for ___")

```typescript
// lib/ai/CategoryDiscovery.ts
export class CategoryDiscovery {
  // ✅ Natural language → category matching
  static async suggestCategories(userQuery: string, context: {
    region?: string;
    userPlan?: string;
    previousCategories?: string[];
  } = {}) {
    // Step 1: Lightweight keyword match
    const keywords = userQuery.toLowerCase().split(/\s+/);
    
    // Step 2: Fetch relevant categories from cache/registry
    const categories = await CategoryRegistryService.getCategories({
      region: context.region,
      tags: keywords
    });
    
    // Step 3: AI semantic ranking (if needed)
    if (categories.length > 10) {
      const ranked = await this.rankWithAI(categories, userQuery, context);
      return ranked.slice(0, 5);
    }
    
    return categories.slice(0, 5);
  }
  
  // ✅ AI ranking with Gemini/OpenAI
  static async rankWithAI(categories: any[], query: string, context: any) {
    try {
      const prompt = `
You are a QR code category matching expert. 
User query: "${query}"
User context: region=${context.region}, plan=${context.userPlan}

Rank these categories by relevance (1=most relevant):
${categories.map((c, i) => `${i+1}. ${c.name}: ${c.description} [tags: ${c.tags.join(', ')}]`).join('\n')}

Return ONLY a JSON array of category IDs in ranked order: ["id1", "id2", ...]
      `;
      
      const response = await fetch('/api/ai/rank-categories', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ prompt, model: 'gemini-1.5-flash' })
      });
      
      const { rankedIds } = await response.json();
      return categories
        .filter(c => rankedIds.includes(c.id))
        .sort((a, b) => rankedIds.indexOf(a.id) - rankedIds.indexOf(b.id));
    } catch {
      // Fallback: simple keyword scoring
      return categories.map(c => ({
        ...c,
        score: keywords.filter(k => 
          c.name.toLowerCase().includes(k) || 
          c.description.toLowerCase().includes(k) ||
          c.tags.some(t => t.includes(k))
        ).length
      })).sort((a, b) => b.score - a.score);
    }
  }
  
  // ✅ "Create custom category" flow for power users
  static async generateCustomCategory(userDescription: string, userId: string) {
    // Step 1: AI extracts schema from natural language
    const schema = await this.extractSchemaFromDescription(userDescription);
    
    // Step 2: Validate + register
    const categoryId = await CategoryRegistryService.registerCategory(schema, userId);
    
    return {
      id: categoryId,
      name: schema.name,
      status: 'pending_approval' // Requires admin review before public visibility
    };
  }
  
  static async extractSchemaFromDescription(description: string) {
    const prompt = `
Extract a QR category schema from this user description:
"${description}"

Return valid JSON matching this TypeScript interface:
{
  "name": string,
  "description": string,
  "fields": [{"key": string, "label": string, "type": string, "required": boolean}],
  "output": {"template": string},
  "tags": string[]
}

Example input: "I need a QR for restaurant table ordering with table number and menu link"
Example output: {
  "name": "Restaurant Table Order",
  "description": "QR for customers to order from their table",
  "fields": [
    {"key": "tableNumber", "label": "Table Number", "type": "number", "required": true},
    {"key": "menuUrl", "label": "Menu URL", "type": "url", "required": true}
  ],
  "output": {"template": "https://order.restaurant.com/table={{tableNumber}}&menu={{menuUrl}}"},
  "tags": ["restaurant", "ordering", "table", "food"]
}
    `;
    
    const response = await fetch('/api/ai/extract-schema', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ prompt })
    });
    
    return await response.json();
  }
}
```
