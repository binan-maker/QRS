### **2. QR Generator Page (`features/generator/`)**

**Problems:**
- **Synchronous QR generation** - Blocking the main thread
- **No caching** - Regenerating same QR codes repeatedly

### **3. My QR Codes Page (`app/my-qr-codes.tsx`)**

**Problems:**
- **Fetching all user QRs at once** - No pagination
- **Full document reads** - Not using field projection
- **No indexing** - Slow queries on large collections
- **Missing React Query caching** - Re-fetching on every navigation

```typescript
// app/qr-detail/[id].tsx - OPTIMIZED VERSION
import { useQuery } from '@tanstack/react-query';
import { doc, getDoc } from 'firebase/firestore';
import { db } from '@/lib/firebase';

// ✅ Use React Query with caching
export function useQrDetail(qrId: string) {
  return useQuery({
    queryKey: ['qr-detail', qrId],
    queryFn: async () => {
      const docRef = doc(db, 'qrCodes', qrId);
      const snap = await getDoc(docRef);
      return { id: snap.id, ...snap.data() };
    },
    staleTime: 1000 * 60 * 5, // Cache for 5 minutes
    gcTime: 1000 * 60 * 30,   // Keep in cache for 30 minutes
  });
}
```

### **Solution 2: Optimize My QR Codes Page**

```typescript
// app/my-qr-codes.tsx - OPTIMIZED VERSION
import { useInfiniteQuery } from '@tanstack/react-query';
import { collection, query, orderBy, limit, startAfter, getDocs } from 'firebase/firestore';
import { db } from '@/lib/firebase';
import { FlashList } from '@shopify/flash-list';

// ✅ Paginated query with field projection
async function getUserQrsPage(uid: string, lastDoc?: any) {
  const baseQuery = query(
    collection(db, 'users', uid, 'generatedQrs'),
    orderBy('createdAt', 'desc'),
    limit(25),
    ...(lastDoc ? [startAfter(lastDoc)] : [])
  );

  const snap = await getDocs(baseQuery);
  return {
    items: snap.docs.map(doc => ({
      id: doc.id,
      // ✅ Only fetch needed fields
      type: doc.data().qrType,
      createdAt: doc.data().createdAt,
      scanCount: doc.data().scanCount || 0,
      status: doc.data().status || 'active',
      thumbnail: doc.data().thumbnailUrl,
    })),
    nextCursor: snap.docs.length === 25 ? snap.docs[snap.docs.length - 1] : null,
  };
}

export function useMyQrs(uid: string) {
  return useInfiniteQuery({
    queryKey: ['my-qrs', uid],
    queryFn: ({ pageParam }) => getUserQrsPage(uid, pageParam),
    getNextPageParam: (lastPage) => lastPage.nextCursor,
    staleTime: 1000 * 60 * 3, // 3 minutes
    gcTime: 1000 * 60 * 60,   // 1 hour
  });
}

export default function MyQRCodesScreen() {

}
```

```typescript
// features/generator/QRGenerator.tsx - OPTIMIZED VERSION
import { useMemo, useCallback, debounce } from 'lodash';
import { useMutation } from '@tanstack/react-query';

export default function QRGenerator() {
  const [content, setContent] = useState('');
  
  // ✅ Debounce content changes (300ms)
  const debouncedContent = useMemo(
    () => debounce((val) => setContent(val), 300),
    []
  );

  // ✅ Memoize QR generation
  const qrImage = useMemo(() => {
    if (!content || content.length < 4) return null;
    return generateQRCodeAsync(content); // Offload to worker
  }, [content]);

  // ✅ Use mutation for saving
  const saveMutation = useMutation({
    mutationFn: saveGeneratedQr,
    onMutate: () => ({ isLoading: true }),
    onSuccess: () => showToast('Saved successfully'),
  });

  const handleSave = useCallback(() => {
    saveMutation.mutate({ content, type: selectedType });
  }, [content, selectedType]);

  return (
    <View>
      <TextInput 
        value={content}
        onChangeText={debouncedContent}
        placeholder="Enter content..."
      />

      <Button 
        title="Save QR" 
        onPress={handleSave}
        disabled={saveMutation.isPending}
      />
    </View>
```
